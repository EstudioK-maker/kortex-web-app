function applySearchEngine(resolvedUrl) {
  const engineId = settings.get('search.engine') || 'google';
  if (engineId === 'google') return resolvedUrl;
  try {
    const u = new URL(resolvedUrl);
    if (/(^|\.)google\.com$/i.test(u.hostname) && u.pathname === '/search' && u.searchParams.has('q')) {
      const build = SEARCH_ENGINES[engineId];
      if (build) return build(u.searchParams.get('q'));
    }
  } catch (e) {}
  return resolvedUrl;
}

function handleKortexRequest(request) {
  const notFound = () => new Response('Not found', { status: 404 });
  let url, rel;
  try { url = new URL(request.url); rel = decodeURIComponent(url.pathname).replace(/^\/+/, ''); } catch (e) { return notFound(); }
  const page = KORTEX_PAGES[url.hostname];
  if (!page) return notFound();
  if (!rel) rel = page;
  const filePath = path.resolve(PAGES_DIR, rel);
  if (!filePath.startsWith(PAGES_DIR + path.sep) || !fs.existsSync(filePath)) return notFound();
  return new Response(fs.readFileSync(filePath), { headers: { 'Content-Type': MIME[path.extname(filePath)] || 'application/octet-stream', 'Cache-Control': 'no-store' } });
}

const INTERNAL_SCHEMES = new Set(['http', 'https', 'kortex', 'about', 'blob', 'data', 'javascript', 'view-source', 'file', 'filesystem', 'chrome', 'chrome-extension', 'devtools', 'ws', 'wss']);

function schemeOf(u) {
  const m = /^([a-z][a-z0-9+.-]*):/i.exec(String(u || '').trim());
  return m ? m[1].toLowerCase() : '';
}
const isInternalUrl = (u) => INTERNAL_SCHEMES.has(schemeOf(u));

const lastExternal = new Map();
function openExternalUrl(url) {
  const scheme = schemeOf(url);
  if (!scheme || isInternalUrl(url)) return;
  const now = Date.now();
  if ((lastExternal.get(url) || 0) > now - 3000) return;
  lastExternal.set(url, now);
  if (lastExternal.size > 100) for (const [k, t] of lastExternal) if (t < now - 10000) lastExternal.delete(k);
  shell.openExternal(url).catch((e) => {
    console.error('[Kortex] Não foi possível abrir ' + scheme + '://', e.message || e);
    toast('Nenhuma aplicação para abrir ' + scheme + '://');
  });
}

function attachNavigationGuards(wc) {
  const guard = (event, url) => {
    try {
      if (!url || isInternalUrl(url)) return;
      event.preventDefault();
      openExternalUrl(url);
    } catch (e) {}
  };
  wc.on('will-frame-navigate', (event, url, isMainFrame) => guard(event, url));
  wc.on('will-redirect', guard);
  wc.on('will-navigate', guard);
}

const isGoogleHost = (u) => {
  try {
    const h = new URL(u).hostname;
    return /(^|\.)google\.com$/i.test(h) || /(^|\.)youtube\.com$/i.test(h);
  } catch (e) { return false; }
};

function watchAuthPopup(wc) {
  wc.on('did-navigate', (_e, url) => {
    if (isGoogleHost(url)) return;
    wc.once('did-finish-load', () => {
      setTimeout(() => {
        try {
          if (!wc.isDestroyed() && !isGoogleHost(wc.getURL())) {
            const win = BrowserWindow.fromWebContents(wc);
            if (win && !win.isDestroyed()) win.close();
          }
        } catch (e) {}
      }, 400);
    });
  });
}

app.on('web-contents-created', (event, contents) => {
  attachNavigationGuards(contents);

  contents.setWindowOpenHandler(({ url: target }) => {
    const gp = googleCompat.popupOptions(target);
    if (gp) return gp;

    if (/^https?:/i.test(target)) {
      createNewTab(target);
      return { action: 'deny' };
    }

    if (!isInternalUrl(target)) {
      openExternalUrl(target);
      return { action: 'deny' };
    }

    return { action: 'allow' };
  });

  contents.on('did-create-window', (win, details) => {
    try { if (googleCompat.isGoogleAuthUrl(details.url)) watchAuthPopup(win.webContents); } catch (e) {}
  });
});

function sendTabsToUI() {
  const list = Object.keys(pageViews).concat(Object.keys(suspended)).map((id) => {
    if (!pageViews[id]) { const s = suspended[id]; return pro.tabInfo(id, { id, title: s.title, url: s.url, isPrivate: s.isPrivate, suspended: true }); }
    const view = pageViews[id];
    const url = view.webContents.getURL();
    let title = (view.webContents.getTitle() || '').trim();
    if (url.startsWith('kortex://')) {
      title = internalTitle(url);
    } else if (url === 'about:blank' || url === '') {
      title = view.isPrivate ? 'Navegação Privada' : 'Novo separador';
    } else if (!title || title === 'K' || title === 'Kortex') {
      try { title = new URL(url).hostname.replace('www.', ''); } catch (e) { title = 'A carregar...'; }
    }
    return pro.tabInfo(id, { id, title, url, isPrivate: view.isPrivate || false });
  });
  list.sort((a, b) => (b.pinned ? 1 : 0) - (a.pinned ? 1 : 0));
  if (chromeView && !chromeView.webContents.isDestroyed()) chromeView.webContents.send('tabs:updated-list', { tabs: list, activeId: activeTabId });
}

function updateTabVisibility() {
  const view = pageViews[activeTabId];
  if (!view) return;
  const url = view.webContents.getURL();
  const isHome = url === 'about:blank' || url === '';
  view.setVisible(!isHome);
  chromeView.webContents.send('show-home', isHome);
}

function applyBounds(view) {
  if (view && pageBounds) view.setBounds(pageBounds);
}

function createNewTab(url = 'about:blank', isPrivate = false) {
  tabCounter++;
  const id = 'tab-' + tabCounter;
  let tabSession = session.defaultSession;
  if (isPrivate) {
    tabSession = session.fromPartition('private-' + Date.now(), { cache: false });
    googleCompat.applySession(tabSession, true);
    tabSession.protocol.handle('kortex', handleKortexRequest);
    pro.setupSession(tabSession);
  }

  const view = new WebContentsView({ webPreferences: { session: tabSession, preload: WEBVIEW_PRELOAD, contextIsolation: true, nodeIntegration: false, sandbox: false, plugins: true } });
  mainWindow.contentView.addChildView(view);
  raiseOverlays();

  pageViews[id] = view;
  view.isPrivate = isPrivate;
  view.setBackgroundColor('#00000000');
  applyBounds(view);
  view.webContents.loadURL(url).catch(() => {});

  pro.attachView(id, view);
  webstore.attachView(view);
  view.webContents.on('dom-ready', () => features.applyZoom(view.webContents));
  view.webContents.on('page-title-updated', () => sendTabsToUI());
  view.webContents.on('did-navigate', () => { updateTabVisibility(); sendTabsToUI(); saveToHistory(view.webContents.getTitle(), view.webContents.getURL()); });
  view.webContents.on('did-finish-load', () => { view.webContents.invalidate(); });

  if (ext && !isPrivate) ext.addTab(view.webContents, mainWindow);
  switchTab(id);
  return id;
}

function switchTab(id) {
  if (suspended[id]) reviveSuspendedTab(id);
  hideSuggestions();
  extMenu.hide();
  if (activeTabId && pageViews[activeTabId]) pageViews[activeTabId].setVisible(false);
  activeTabId = id;
  if (pageViews[id]) { applyBounds(pageViews[id]); updateTabVisibility(); pageViews[id].webContents.focus(); }
  sendTabsToUI();
  markActive(id);
}

function closeTab(id) {
  if (pageViews[id]) {
    pro.onClose(pageViews[id]);
    if (ext && !pageViews[id].isPrivate) ext.removeTab(pageViews[id].webContents);
    mainWindow.contentView.removeChildView(pageViews[id]);
    pageViews[id].webContents.close();
    delete pageViews[id];
  }
  if (activeTabId === id) {
    const rest = Object.keys(pageViews);
    if (rest.length > 0) switchTab(rest[rest.length - 1]); else createNewTab();
  }
  sendTabsToUI();
}

function navigateActive(url) {
  const view = pageViews[activeTabId];
  if (!view) return;
  view.webContents.loadURL(url).catch(() => {});
  updateTabVisibility();
}

ipcMain.handle('tabs:reorder', (e, ids) => {
  if (!Array.isArray(ids)) return false;
  const known = Object.keys(pageViews);
  const order = ids.filter((id) => known.includes(id));
  if (order.length !== known.length || new Set(order).size !== order.length) return false;
  const entries = order.map((id) => [id, pageViews[id]]);
  known.forEach((id) => delete pageViews[id]);
  entries.forEach(([id, v]) => { pageViews[id] = v; });
  sendTabsToUI();
  return true;
});

ipcMain.on('suggest:show', (e, payload) => {
  if (e.sender !== chromeView.webContents) return;
  const { items, selected, rect } = payload || {};
  if (!Array.isArray(items) || !items.length || !rect) { hideSuggestions(); return; }
  const height = Math.min(items.length * SUGGEST_ROW_H + 12, SUGGEST_MAX_H);
  suggestView.setBounds({ x: Math.round(rect.x), y: Math.round(rect.y + rect.height + 4), width: Math.round(rect.width), height });
  mainWindow.contentView.addChildView(suggestView);
  mainWindow.contentView.addChildView(overlayView);
  suggestView.webContents.send('suggest:render', { items: items.map(normalizeSuggestion), selected: Number.isInteger(selected) ? selected : -1 });
  suggestView.setVisible(true);
});

ipcMain.on('suggest:pick', (e, item) => {

}

function splitCsvLine(line) {
  const out = []; let cur = '', inQ = false;
  for (let i = 0; i < line.length; i++) {
    const c = line[i];
    if (inQ) { if (c === '"') { if (line[i + 1] === '"') { cur += '"'; i++; } else inQ = false; } else cur += c; }
    else if (c === '"') inQ = true;
    else if (c === ',') { out.push(cur); cur = ''; }
    else cur += c;
  }
  out.push(cur);
  return out;
}
function parseChromePasswordsCsv(text) {
  const lines = text.split(/\r?\n/).filter(Boolean);
  if (!lines.length) return [];
  const header = splitCsvLine(lines[0]).map((h) => h.trim().toLowerCase());
  const urlIdx = header.indexOf('url'), userIdx = header.indexOf('username'), passIdx = header.indexOf('password');
  if (urlIdx < 0 || passIdx < 0) return [];
  const out = [];
  for (let i = 1; i < lines.length; i++) {
    const cols = splitCsvLine(lines[i]);
    const url = cols[urlIdx], password = cols[passIdx];
    if (!url || !password) continue;
    try { out.push({ domain: new URL(url).hostname, username: userIdx >= 0 ? cols[userIdx] : '', password }); } catch (e) {}
  }
  return out;
}

let bookmarks = null;
const bm = () => (bookmarks = bookmarks || readJson('bookmarks.json', []));
function toggleBookmark(url, title) {
  if (!/^(https?:|kortex:)/i.test(url || '')) return { ok: false };
  const list = bm(); const i = list.findIndex((b) => b.url === url);
  if (i > -1) list.splice(i, 1); else list.push({ id: crypto.randomUUID(), url, title: title || url, added: Date.now() });
  writeJson('bookmarks.json', list); toChrome('bookmarks:changed');
  return { ok: true, bookmarked: i === -1 };
}
function parseNetscapeBookmarks(html) {
  const out = [];
  const re = /<A[^>]*HREF="([^"]+)"[^>]*>([^<]*)<\/A>/gi;
  let m;
  while ((m = re.exec(html))) {
    const url = m[1].trim();
    const title = (m[2] || '').trim() || url;
    if (/^https?:\/\//i.test(url)) out.push({ url, title });
  }
  return out;
}

function setupImports() {
  ipcMain.handle('import:bookmarks', async () => {
    const { canceled, filePaths } = await dialog.showOpenDialog(ctx.getMainWindow(), {
      properties: ['openFile'], filters: [{ name: 'Favoritos (HTML)', extensions: ['html', 'htm'] }]
    });
    if (canceled || !filePaths[0]) return { success: false };
    try {
      const html = fs.readFileSync(filePaths[0], 'utf8');
      const imported = parseNetscapeBookmarks(html);
      const list = bm();
      let added = 0;
      for (const b of imported) {
        if (!list.find((x) => x.url === b.url)) {
          list.push({ id: crypto.randomUUID(), url: b.url, title: b.title, added: Date.now() });
          added++;
        }
      }
      writeJson('bookmarks.json', list);
      toChrome('bookmarks:changed');
      return { success: true, imported: added, total: imported.length };
    } catch (err) {
      return { success: false, error: err.message };
    }
  });
}

function describePermission(permission, details) {
  switch (permission) {
    case 'media': { const t = (details && details.mediaTypes) || []; return t.includes('video') && t.includes('audio') ? 'usar a câmara e o microfone' : t.includes('video') ? 'usar a câmara' : 'usar o microfone'; }

  }
}
async function askPermission(origin, permission, details, k, store, persist) {
  let host = origin; try { host = new URL(origin).host; } catch (e) {}
  const r = await dialog.showMessageBox({ type: 'question', buttons: ['Permitir', 'Bloquear'], defaultId: 1, cancelId: 1, title: 'Pedido de permissão', message: `${host} quer ${describePermission(permission, details)}`, checkboxLabel: persist ?   const allowed = r.response === 0; if (r.checkboxChecked) { store[k] = allowed; if (persist) writeJson('permissions.json', store); } return allowed;

function setupPermissions(ses, { persist = true } = {}) {
  const store = persist ? (perms = perms || readJson('permissions.json', {})) : {};
  const pendingHere = new Map(); const key = (origin, p) => origin + '|' + p;
  ses.setPermissionCheckHandler((wc, permission, origin) => AUTO_ALLOW.has(permission) || store[key(origin, permission)] === true);
  ses.setPermissionRequestHandler(async (wc, permission, callback, details) => {
    if (AUTO_ALLOW.has(permission)) return callback(true); if (!ASK.has(permission)) return callback(false);
    let origin = ''; try { origin = new URL(details.requestingUrl || wc.getURL()).origin; } catch (e) {}
    if (!origin || origin === 'null') return callback(false);
    const k = key(origin, permission); if (k in store) return callback(store[k]); if (pendingHere.has(k)) return pendingHere.get(k).then(callback);
    const p = askPermission(origin, permission, details, k, store, persist); pendingHere.set(k, p); let allowed = false;
    try { allowed = await p; } finally { pendingHere.delete(k); } callback(allowed);
  });
}

function attach(wc) {
  wc.on('before-input-event', (event, input) => {
    if (input.type !== 'keyDown') return;
    const ctrl = input.control || input.meta; const key = (input.key || '').toLowerCase();
    const run = (fn) => { event.preventDefault(); try { fn(); } catch (e) {} };
    const v = active(); const awc = v && v.webContents;

    if (key === 'escape') { toChrome('find:close'); return; }
    if (key === 'f12' || (ctrl && input.shift && key === 'i')) return run(() => awc && awc.toggleDevTools());
    if (key === 'f11') return run(() => { const w = ctx.getMainWindow(); w.setFullScreen(!w.isFullScreen()); });
    if (key === 'f5') return run(() => awc && awc.reload());
    if (input.alt && !ctrl && (key === 'arrowleft' || key === 'arrowright')) { return run(() => { if (!awc) return; const h = navOf(awc); if (key === 'arrowleft') { if (h.canGoBack()) h.goBack(); } else if (h.canGoForward()) h.goForward(); }); }
    if (!ctrl) return;

    if (key === '+' || key === '=') return run(() => zoom(0.1));
    if (key === '-') return run(() => zoom(-0.1));
    if (key === '0') return run(() => zoom(0, true));

    if (input.shift) {
      if (key === 't') return run(reopenClosed);
      if (key === 'n') return run(() => ctx.createNewTab('about:blank', true));
      if (key === 'a') return run(() => toChrome('assistant:toggle'));
      if (key === 'k') return run(() => { toChrome('tabsearch:open'); ctx.getChromeView().webContents.focus(); });
      if (key === 'b') return run(() => toChrome('bookmarks:toggle-bar'));
      if (key === 'r') return run(() => awc && awc.reloadIgnoringCache());
      if (key === 'tab') return run(() => cycle(-1));
      return;
    }

    if (key === 't') return run(() => ctx.createNewTab());
    if (key === 'w') return run(() => ctx.closeTab(ctx.getActiveId()));
    if (key === 'tab') return run(() => cycle(1));
    if (key === 'r') return run(() => awc && awc.reload());
    if (key === 'p') return run(() => awc && awc.print());
    if (key === 'u') return run(() => awc && ctx.createNewTab('view-source:' + awc.getURL()));
    if (key === 'h') return run(() => openInternal('kortex://history'));
    if (key === 'j') return run(() => openInternal('kortex://downloads'));

  });
}

const SECRET_KEYS = ['ai.apiKey'];

let store = {};
let needsMigration = false;
const listeners = [];
const clone = (x) => JSON.parse(JSON.stringify(x === undefined ? null : x));

function isValid(def, v) {
  switch (def.type) {
    case 'bool':   return typeof v === 'boolean';
    case 'enum':   return def.values.includes(v);
    case 'string': return typeof v === 'string' && (!def.validate || def.validate(v));
    case 'array':  return Array.isArray(v) && (!def.validate || def.validate(v));
    case 'object': return !!v && typeof v === 'object' && !Array.isArray(v) && (!def.validate || def.validate(v));
    default:       return false;
  }
}

function load() {
  try {
    if (fs.existsSync(SETTINGS_PATH)) store = JSON.parse(fs.readFileSync(SETTINGS_PATH, 'utf8')) || {};
  } catch (e) {
    console.warn('[Kortex Settings] settings.json ilegível, a usar predefinições:', e.message);
    store = {};
  }
  for (const key in SCHEMA) {
    const def = SCHEMA[key];
    if (def.type === 'action') { delete store[key]; continue; }
    if (SECRET_KEYS.includes(key)) {
      const { value, wasPlain } = secrets.decrypt(store[key]);
      store[key] = value;
      if (wasPlain && value && secrets.available()) needsMigration = true; // havia um valor em texto simples: regravar cifrado
      continue;
    }
    if (!(key in store) || !isValid(def, store[key])) store[key] = clone(def.default);
  }
  for (const key of SECRET_KEYS) if (typeof store[key] !== 'string') store[key] = '';
}

function save() {
  const out = { ...store };
  for (const key of SECRET_KEYS) {
    const enc = secrets.encrypt(store[key]);
    out[key] = enc === null ? '' : enc; // sem cifra disponível, é preferível não guardar a chave a guardá-la em claro
  }
  const tmp = SETTINGS_PATH + '.tmp';
  fs.writeFileSync(tmp, JSON.stringify(out, null, 2));
  fs.renameSync(tmp, SETTINGS_PATH);
}

function init() {
  load();
  if (needsMigration) { try { save(); console.log('[Kortex Settings] chave da API migrada para armazenamento cifrado'); } catch (e) { console.warn('[Kortex Settings] migração falhou:', e.message); } }
  for (const key in SCHEMA) {
    const def = SCHEMA[key];
    if (def.apply && def.type !== 'action') {
      try { def.apply(store[key]); } catch (e) { console.warn(`[Kortex Settings] apply(${key}):`, e.message); }
    }
  }
}

  const out = {};

  return out;
}

async function set(key, value) {
  const def = SCHEMA[key];
  if (!def) throw new Error(`Definição desconhecida: ${key}`);
  if (def.type === 'action') { await def.apply?.(); return { success: true }; }
  if (!isValid(def, value)) throw new Error(`Valor inválido para ${key}`);
  if (JSON.stringify(store[key]) === JSON.stringify(value)) return { success: true, changed: false };

  store[key] = value;
  save();
  try { def.apply?.(value); } catch (e) { console.warn(`[Kortex Settings] apply(${key}):`, e.message); }
  for (const cb of listeners) { try { cb(key, value); } catch (e) { /* um listener com erro não bloqueia os outros */ } }
  return { success: true, changed: true, restart: !!def.restart, reloadTabs: !!def.reloadTabs };
}

async function reset(prefix = '') {
  let changed = 0;
  for (const key in SCHEMA) {
    const def = SCHEMA[key];
    if (def.type === 'action' || def.noReset || !key.startsWith(prefix)) continue;
    const r = await set(key, clone(def.default));
    if (r.changed) changed++;
  }
  return { success: true, changed };
}

const onChanged = (cb) => { listeners.push(cb); };

  const tpl = ENGINES[store['search.engine']] || ENGINES.google;
  return tpl.replace('%s', () => encodeURIComponent(query));
}

function resolveInput(text) {
  const t = String(text || '').trim();
  if (!t) return 'about:blank';
  if (/^(https?|file|ftp):\/\//i.test(t) || /^(about|kortex|view-source):/i.test(t)) return t;
  if (/^localhost(:\d+)?([/?#]|$)/i.test(t) || /^\d{1,3}(\.\d{1,3}){3}(:\d+)?([/?#]|$)/.test(t)) return 'http://' + t;
  if (!/\s/.test(t) && /^[^\s/?#]+\.[a-z]{2,}(:\d+)?([/?#]\S*)?$/i.test(t)) return 'https://' + t;
  return searchUrl(t);
}

const hostOf = u => { try { return new URL(u).hostname.toLowerCase(); } catch { return ''; } };
function isCookieException(host) {
  const list = settings.get('privacy.cookieExceptions') || [];
  return list.some(raw => {
    const e = String(raw).trim().toLowerCase().replace(/^[a-z]+:\/\//, '').replace(/^\[\*\.\]/, '').replace(/\/.*$/, '');
    return e && (host === e || host.endsWith('.' + e));
  });
}
function isThirdParty(details) {
  if (details.resourceType === 'mainFrame') return false;
  const wc = details.webContentsId ? webContents.fromId(details.webContentsId) : null;
  if (!wc || wc.isDestroyed()) return false;
  const pageHost = hostOf(wc.getURL());
  const reqHost = hostOf(details.url);
  if (!pageHost || !reqHost || isCookieException(pageHost)) return false;
  return getDomain(reqHost) !== getDomain(pageHost);
}
const deleteHeader = (headers, name) => { for (const k of Object.keys(headers)) if (k.toLowerCase() === name) delete headers[k]; };

function setupWebRequest() {
  const ses = session.defaultSession;
  ses.webRequest.onBeforeSendHeaders((details, callback) => {
    const headers = details.requestHeaders;
    if (settings.get('privacy.dnt')) headers['DNT'] = '1';
    if (settings.get('privacy.gpc')) headers['Sec-GPC'] = '1';
    if (settings.get('privacy.blockThirdPartyCookies') && isThirdParty(details)) deleteHeader(headers, 'cookie');
    callback({ requestHeaders: headers });
  });
  ses.webRequest.onHeadersReceived((details, callback) => {
    const headers = details.responseHeaders;
    if (settings.get('privacy.blockThirdPartyCookies') && isThirdParty(details)) deleteHeader(headers, 'set-cookie');
    callback({ responseHeaders: headers });
  });
}

function uniquePath(dir, name) {
  const safe = path.basename(name);
  const ext = path.extname(safe);
  const base = path.basename(safe, ext);
  let p = path.join(dir, safe);
  let n = 1;
  while (fs.existsSync(p)) p = path.join(dir, `${base} (${n++})${ext}`);
  return p;
}

function setupDownloads() {

}

function init({ settings, getActiveView, getAllViews, getSkillsPrompt }) {
  const controllers = new Map();

  async function readPage(view, max) {
    if (!view || view.isPrivate || view.webContents.isDestroyed()) return null;
    const url = view.webContents.getURL();
    if (!/^https?:/i.test(url)) return null;
    try {
      const code = `JSON.stringify({t:document.title,s:String(getSelection()).slice(0,4000),x:((document.body&&document.body.innerText)||'').slice(0,${max})})`;
      const raw = await Promise.race([
        view.webContents.executeJavaScript(code),
        new Promise((r) => setTimeout(() => r(null), 3000))
      ]);
      return raw ? { url, ...JSON.parse(raw) } : null;
    } catch { return null; }
  }

  async function buildContext({ withPage, withTabs }) {
    if (settings.get('assistant.pageAccess') === false) return '';
    if (withTabs) {
      const views = Object.values(getAllViews()).slice(0, 8);
      const pages = (await Promise.all(views.map((v) => readPage(v, 3000)))).filter(Boolean);
      if (!pages.length) return '';
      return '\n\n[SEPARADORES ABERTOS]\n' + pages.map((p, i) => `(${i + 1}) ${p.t}\n${p.url}\n"""${p.x}"""`).join('\n\n');

function crxToZip(buf) {
  if (buf.length > 16 && buf.toString('latin1', 0, 4) === 'Cr24') {
    const ver = buf.readUInt32LE(4);
    if (ver === 3) return buf.subarray(12 + buf.readUInt32LE(8));
    if (ver === 2) return buf.subarray(16 + buf.readUInt32LE(8) + buf.readUInt32LE(12));
    throw new Error('Versao de CRX desconhecida: ' + ver);
  }
  if (buf.length > 4 && buf[0] === 0x50 && buf[1] === 0x4B) return buf;
  throw new Error('O ficheiro descarregado nao e uma extensao valida.');
}

// src/main/secrets.js — cifra de segredos (palavras-passe, chave da API) com o safeStorage do Electron.
// No Windows usa a DPAPI: só o mesmo utilizador do Windows, neste PC, consegue decifrar.
// Formato guardado: "enc:v1:<base64>". Texto sem este prefixo é tratado como valor antigo em texto simples (migração).
const { safeStorage } = require('electron');

const PREFIX = 'enc:v1:';

const available = () => { try { return safeStorage.isEncryptionAvailable(); } catch { return false; } };
const isEncrypted = (v) => typeof v === 'string' && v.startsWith(PREFIX);

// Devolve o valor cifrado; se a cifra não estiver disponível devolve null (quem chama decide o que fazer).
function encrypt(plain) {
  if (plain === '' || plain == null) return '';
  if (!available()) return null;
  return PREFIX + safeStorage.encryptString(String(plain)).toString('base64');
}

// Devolve { value, wasPlain }. Valores que não se conseguem decifrar (outro utilizador/PC) dão ''.
function decrypt(stored) {
  if (stored === '' || stored == null) return { value: '', wasPlain: false };
  if (!isEncrypted(stored)) return { value: String(stored), wasPlain: true };
  try { return { value: safeStorage.decryptString(Buffer.from(stored.slice(PREFIX.length), 'base64')), wasPlain: false }; }
  catch { return { value: '', wasPlain: false }; }
}

module.exports = { encrypt, decrypt, isEncrypted, available };

  setInterval(() => { if (!blocking) check(); }, opts.intervalMs);
}

    for (const p of PROTOCOLS) { try { ok = app.setAsDefaultProtocolClient(p) && ok; } catch (e) { ok = false; } }
  }
  return { ok, needsUserAction: win };
}

async function offerOnFirstRun({ getWindow, canPrompt } = {}) {
  if (process.env.KORTEX_RESET_DEFAULT_PROMPT) resetPrompt();
  if (readState().defaultBrowserAsked) { log('já foi perguntado antes: a saltar'); return; }
  if (isDefault()) { log('o Kortex já é o predefinido'); writeState({ defaultBrowserAsked: true }); return; }
  if (!app.isPackaged && !process.env.KORTEX_FORCE_DEFAULT_PROMPT) { log('modo dev: pergunta desativada'); return; }

  for (let i = 0; i < 30 && typeof canPrompt === 'function' && !canPrompt(); i++) await sleep(2000);

  const win = process.platform === 'win32';
  const mac = process.platform === 'darwin';
  const opts = {
    type: 'question', title: 'Kortex', message: 'Tornar o Kortex o teu navegador predefinido?',
    detail: win ? 'O Windows vai abrir as Definições: em "Navegador web" escolhe Kortex.' : (mac ? 'O macOS vai pedir-te para confirmar a mudança.' : 'Os links de outras aplicações passam a abrir no Kortex.'),
    buttons: ['Tornar predefinido', 'Agora não'], defaultId: 0, cancelId: 1, noLink: true
  };
  const parent = typeof getWindow === 'function' ? getWindow() : null;
  try { if (parent && !parent.isDestroyed()) { parent.show(); parent.focus(); } } catch (e) {}
  let r;
  try { r = await (parent && !parent.isDestroyed() ? dialog.showMessageBox(parent, opts) : dialog.showMessageBox(opts)); }
  catch (e) { r = await dialog.showMessageBox(opts); }
  writeState({ defaultBrowserAsked: true });
  log('resposta:', r.response === 0 ? 'tornar predefinido' : 'agora não');
  if (r.response === 0) log('resultado:', JSON.stringify(await makeDefault({ parentWindow: parent })));
}

function isSafeWebUrl(u) { try { const p = new URL(u).protocol; return p === 'http:' || p === 'https:'; } catch (e) { return false; } }
function fileToUrl(p) { try { const abs = path.resolve(p); if (FILE_EXTS.has(path.extname(abs).toLowerCase()) && fs.statSync(abs).isFile()) return pathToFileURL(abs).href; } catch (e) {} return null; }
function urlFromArgv(argv) {
  for (const a of (argv || []).slice(1)) {
    if (!a || a.startsWith('-')) continue;
    if (isSafeWebUrl(a)) return a;
    const f = fileToUrl(a); if (f) return f;
  }
  return null;
}

let openUrlFn = null;
const pending = [];
function deliver(url) { if (!url) return; if (openUrlFn) openUrlFn(url); else pending.push(url); }

app.on('open-url', (e, url) => { e.preventDefault(); if (isSafeWebUrl(url)) deliver(url); });
app.on('open-file', (e, filePath) => { e.preventDefault(); deliver(fileToUrl(filePath)); });

function init({ openUrl, getWindow, canPrompt, handleSecondInstance = false, handleColdStart = false, offerDelayMs = 1500 } = {}) {
  if (typeof openUrl !== 'function') throw new Error('default-browser: init({ openUrl }) é obrigatório');
  openUrlFn = openUrl;
  log('init | instalada:', app.isPackaged, '| já é predefinido:', isDefault());
  if (handleSecondInstance) app.on('second-instance', (_e, argv) => deliver(urlFromArgv(argv)));
  if (handleColdStart) deliver(urlFromArgv(process.argv));
  while (pending.length) openUrlFn(pending.shift());
  setTimeout(() => { offerOnFirstRun({ getWindow, canPrompt }).catch((e) => log('erro na pergunta:', e && e.message)); }, offerDelayMs);
}

'use strict';

const CONTRAST_FIX_HOSTS = ['workspace.google.com', 'mail.google.com'];

function contrastFix() {
  var KEY = Symbol.for('kx.contrast');
  if (document[KEY]) return document[KEY]();

  var mq = window.matchMedia('(prefers-color-scheme: dark)');
  var LIGHT = '#e8eaed';
  var SKIP = { SCRIPT: 1, STYLE: 1, NOSCRIPT: 1, TEMPLATE: 1 };
  var COLOR_RE = /(rgba?|color|lab|lch|oklab|oklch|hsla?|hwb)\([^)]*\)/g;
  var fixed = new Map();

  var cx = document.createElement('canvas').getContext('2d', { willReadFrequently: true });
  cx.canvas.width = 1; cx.canvas.height = 1;
  function parse(c) {
    if (!c) return null;
    if (c === 'transparent') return { r: 0, g: 0, b: 0, a: 0 };
    var m = /^rgba?\(([^)]+)\)$/.exec(c);
    if (m) {
      var p = m[1].split(/[\s,\/]+/).filter(Boolean).map(parseFloat);
      return { r: p[0], g: p[1], b: p[2], a: p.length > 3 ? p[3] : 1 };
    }
    try {
      cx.clearRect(0, 0, 1, 1);
      cx.fillStyle = '#000'; cx.fillStyle = c;
      cx.fillRect(0, 0, 1, 1);
      var d = cx.getImageData(0, 0, 1, 1).data;
      return { r: d[0], g: d[1], b: d[2], a: d[3] / 255 };
    } catch (e) { return null; }
  }
  function lin(v) { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); }
  function lum(c) { return 0.2126 * lin(c.r) + 0.7152 * lin(c.g) + 0.0722 * lin(c.b); }
  function ratio(a, b) {
    var l1 = lum(a), l2 = lum(b);
    if (l1 < l2) { var t = l1; l1 = l2; l2 = t; }
    return (l1 + 0.05) / (l2 + 0.05);
  }
  function over(f, b) {
    var a = f.a;
    return { r: f.r * a + b.r * (1 - a), g: f.g * a + b.g * (1 - a), b: f.b * a + b.b * (1 - a), a: 1 };
  }
  function clipsText(cs) { return (cs.webkitBackgroundClip || cs.backgroundClip) === 'text'; }

  function up(n) {
    if (n.parentElement) return n.parentElement;
    var r = n.getRootNode && n.getRootNode();
    return r && r.host ? r.host : null;
  }

  function bgOf(el, skipSelf) {
    var layers = [], base = null;
    for (var n = skipSelf ? up(el) : el; n && n.nodeType === 1; n = up(n)) {
      var cs = getComputedStyle(n);
      var c = parse(cs.backgroundColor);
      if (c && c.a >= 0.95) { base = c; break; }
      if (c && c.a > 0) layers.push(c);
      if (cs.backgroundImage && cs.backgroundImage !== 'none') {
        var stops = (cs.backgroundImage.match(COLOR_RE) || []).map(parse).filter(Boolean);
        var solid = stops.filter(function (s) { return s.a >= 0.95; });
        if (solid.length) {
          var sum = { r: 0, g: 0, b: 0 };
          solid.forEach(function (s) { sum.r += s.r; sum.g += s.g; sum.b += s.b; });
          base = { r: sum.r / solid.length, g: sum.g / solid.length, b: sum.b / solid.length, a: 1 };
          break;
        }
      }
    }
    if (!base) base = { r: 32, g: 33, b: 36, a: 1 };
    for (var i = layers.length - 1; i >= 0; i--) base = over(layers[i], base);
    return base;
  }

  function fgOf(cs, bg) {
    var fill = parse(cs.webkitTextFillColor);
    if (clipsText(cs) || (fill && fill.a === 0)) {
      var stops = (cs.backgroundImage.match(COLOR_RE) || []).map(parse).filter(Boolean);
      if (!stops.length) return null;
      var best = over(stops[0], bg);
      for (var i = 1; i < stops.length; i++) {
        var s = over(stops[i], bg);
        if (lum(s) > lum(best)) best = s;
      }
      return best;
    }
    var c = (fill && fill.a > 0) ? fill : parse(cs.color);
    if (!c || c.a === 0) return null;
    return over(c, bg);
  }

  function collect(root, out) {
    var w = document.createTreeWalker(root, NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT);
    var node, count = 0;
    while ((node = w.nextNode()) && count++ < 40000) {
      if (node.nodeType === 1) { if (node.shadowRoot) collect(node.shadowRoot, out); continue; }
      var t = node.nodeValue;
      if (!t || !t.trim()) continue;
      var p = node.parentElement;
      if (!p || SKIP[p.tagName]) continue;
      out.add(p);
    }
  }

  function put(el, prop, v) { if (v) el.style.setProperty(prop, v); else el.style.removeProperty(prop); }
  function restore() {
    fixed.forEach(function (orig, el) {
      put(el, 'color', orig[0]);
      put(el, '-webkit-text-fill-color', orig[1]);
    });
    fixed.clear();
  }

  function run() {
    if (!mq.matches) { restore(); return { fixed: 0, scanned: 0, dark: false }; }
    fixed.forEach(function (_, el) { if (!el.isConnected) fixed.delete(el); });

    var parents = new Set();
    collect(document.documentElement, parents);

    var scanned = 0;
    parents.forEach(function (el) {
      if (fixed.has(el)) return;
      scanned++;
      var cs = getComputedStyle(el);
      if (cs.display === 'none' || cs.visibility === 'hidden' || cs.opacity === '0') return;
      var bg = bgOf(el, clipsText(cs));
      if (lum(bg) > 0.2) return;
      var fg = fgOf(cs, bg);
      if (!fg) return;
      var googleDark = Math.round(fg.r) === 32 && Math.round(fg.g) === 33 && Math.round(fg.b) === 36;
      if (googleDark || ratio(fg, bg) < 3) {
        fixed.set(el, [el.style.getPropertyValue('color'), el.style.getPropertyValue('-webkit-text-fill-color')]);
        el.style.setProperty('color', LIGHT, 'important');
        el.style.setProperty('-webkit-text-fill-color', LIGHT, 'important');
      }
    });
    return { fixed: fixed.size, scanned: scanned, dark: true };
  }

  var timer = null;
  function schedule() { clearTimeout(timer); timer = setTimeout(run, 150); }

  Object.defineProperty(document, KEY, { value: run, configurable: true });
  new MutationObserver(schedule).observe(document.documentElement, {
    childList: true, subtree: true, attributes: true, attributeFilter: ['class']
  });
  if (mq.addEventListener) mq.addEventListener('change', run);
  window.addEventListener('load', schedule, { once: true });
  window.addEventListener('scroll', schedule, { passive: true });
  [500, 1500, 3000].forEach(function (ms) { setTimeout(run, ms); });
  return run();
}

function applyContrastFix(wc) {
  try {
    if (!wc || wc.isDestroyed()) return;
    const host = new URL(wc.getURL()).hostname;
    if (!CONTRAST_FIX_HOSTS.includes(host)) return;
    wc.executeJavaScript('(' + contrastFix.toString() + ')()')
      .then((r) => { if (process.env.KORTEX_DEBUG) console.log('[contrastFix]', host, r); })
      .catch((e) => { if (process.env.KORTEX_DEBUG) console.error('[contrastFix]', e); });
  } catch (e) {}
}
