# CONTEXTO DO PROJETO: KORTEX

Olá. Estou a desenvolver o **Kortex**, um navegador web baseado em Electron (com WebContentsView). O projeto tem uma interface customizada (HTML/CSS/JS no renderer), suporte a extensões e um sistema de IA. 

O código já sofreu várias alterações por outras IAs (Claude e ChatGPT) e o ficheiro main.js ficou com "remendos", erros de sintaxe e lógicas conflituosas.

## OS PROBLEMAS QUE PRECISAM DE SER RESOLVIDOS:
1. **Bloqueio do Google (Gmail e Meet):** O Google bloqueia o login e mostra a barra azul *"This browser version is no longer supported"*. Preciso de uma implementação limpa que force o User-Agent e os Client Hints (sec-ch-ua) de um Chrome real na session.defaultSession. E que injete um preload (chrome-spoof.js) para fingir o window.chrome e 
avigator.userAgentData em todas as páginas.
2. **Popups de Login (Comet):** Quando clico em "Login com Google" num site (ex: Perplexity), o popup abre no meu navegador externo (Comet). Preciso que o setWindowOpenHandler intercete os links do ccounts.google.com e os abra dentro do Kortex (numa BrowserWindow nativa com o chrome-spoof.js). Para links normais (http/https), deve abrir num novo separador do Kortex.
3. **Partilha de Ecrã (Meet):** A partilha de ecrã não funciona. Preciso de usar o session.setDisplayMediaRequestHandler com o desktopCapturer para abrir a janela de seleção nativa do Windows.
4. **Protocolos Externos (Roblox):** Links como oblox:// não abrem a app. Preciso de intercetar todos os protocolos que não sejam http, https ou kortex e usar o shell.openExternal(url) para abrir a aplicação no Windows.
5. **Navegador Predefinido:** O Windows não mostra o Kortex nas opções de navegador predefinido. Preciso de configurar o uild.protocols no package.json para http e https, e adicionar o pp.requestSingleInstanceLock() no main.js para apanhar links quando a app já está aberta.

function handleKortexRequest(request) {
  const notFound = () => new Response('Not found', { status: 404 });
  let url, rel;
  try { url = new URL(request.url); rel = decodeURIComponent(url.pathname).replace(/^\/+/, ''); } catch (e) { return notFound(); }
  const page = KORTEX_PAGES[url.hostname];
  if (!page) return notFound();
  if (!rel) rel = page;
  const filePath = path.resolve(PAGES_DIR, rel);
  if (!filePath.startsWith(PAGES_DIR + path.sep) || !fs.existsSync(filePath)) return notFound();

  return new Response(fs.readFileSync(filePath), { headers: { 'Content-Type': MIME[path.extname(filePath)] || 'application/octet-stream', 'Cache-Control': 'no-store' } });
}

      return { action: 'deny' };
    }

    // 4. Qualquer outra coisa, bloqueia
    return { action: 'deny' };
  });
});

function sendTabsToUI() {
  const list = Object.keys(pageViews).map((id) => {
    const view = pageViews[id];
    const url = view.webContents.getURL();
    let title = (view.webContents.getTitle() || '').trim();
    if (url.startsWith('kortex://')) {
      title = internalTitle(url);
    } else if (url === 'about:blank' || url === '') {
      title = view.isPrivate ? 'NavegaÃ§Ã£o Privada' : 'Novo separador';
    } else if (!title || title === 'K' || title === 'Kortex') {
      try { title = new URL(url).hostname.replace('www.', ''); } catch (e) { title = 'A carregar...'; }
    }
    return pro.tabInfo(id, { id, title, url, isPrivate: view.isPrivate || false });
  });
  list.sort((a, b) => (b.pinned ? 1 : 0) - (a.pinned ? 1 : 0));
  if (chromeView && !chromeView.webContents.isDestroyed()) chromeView.webContents.send('tabs:updated-list', { tabs: list, activeId: activeTabId });
}

function updateTabVisibility() {
  const view = pageViews[activeTabId];
  if (!view) return;
  const url = view.webContents.getURL();
  const isHome = url === 'about:blank' || url === '';
  view.setVisible(!isHome);
  chromeView.webContents.send('show-home', isHome);
}

function createNewTab(url = 'about:blank', isPrivate = false) {
  tabCounter++;
  const id = 'tab-' + tabCounter;
  let tabSession = session.defaultSession;
  if (isPrivate) {
    tabSession = session.fromPartition('private-' + Date.now(), { cache: false });
    googleCompat.applySession(tabSession, true);
    tabSession.protocol.handle('kortex', handleKortexRequest);
  }

  const view = new WebContentsView({ webPreferences: { session: tabSession, preload: WEBVIEW_PRELOAD, contextIsolation: true, nodeIntegration: false, sandbox: false } });
  view.webContents.session.setPreloads([SPOOF_PRELOAD, WEBVIEW_PRELOAD]);
  mainWindow.contentView.addChildView(view);
  raiseOverlays();

  pageViews[id] = view;
  view.isPrivate = isPrivate;
  view.setBackgroundColor('#00000000');
  applyBounds(view);
  view.webContents.loadURL(url).catch(() => { /* falhas de carregamento tratadas em browser-features (pÃ¡gina de erro) */ });
  // Interceta qualquer tentativa de abrir uma app externa (roblox://, discord://, etc.)
  view.webContents.on('will-navigate', (event, url) => {
    if (!url.startsWith('http') && !url.startsWith('kortex')) {
      event.preventDefault();
      require('electron').shell.openExternal(url).catch(e => console.error('Erro ao abrir app:', e));
    }
  });

  // Interceta redirecionamentos
  view.webContents.on('will-redirect', (event, url) => {
    if (!url.startsWith('http') && !url.startsWith('kortex')) {
      event.preventDefault();
      require('electron').shell.openExternal(url).catch(e => console.error('Erro ao abrir app:', e));
    }
  });


  pro.attachView(id, view);
  webstore.attachView(view);
  if (isPrivate) pro.setupSession(tabSession);
  view.webContents.on('dom-ready', () => features.applyZoom(view.webContents));

  
  


  view.webContents.on('page-title-updated', () => sendTabsToUI());
  view.webContents.on('did-navigate', () => { updateTabVisibility(); sendTabsToUI(); saveToHistory(view.webContents.getTitle(), view.webContents.getURL()); });
  view.webContents.on('did-finish-load', () => { view.webContents.invalidate(); });

  if (ext && !isPrivate) ext.addTab(view.webContents, mainWindow);
  switchTab(id);
  return id;
}

  if (pageViews[id]) {
    pro.onClose(pageViews[id]);
    if (ext && !pageViews[id].isPrivate) ext.removeTab(pageViews[id].webContents);

    if (rest.length > 0) switchTab(rest[rest.length - 1]); else createNewTab();

  sendTabsToUI();
}

// MigraÃ§Ã£o: palavras-passe antigas (texto simples) passam a cifradas no arranque.
function migrateAutofill() {
  const db = readAutofill();
  if (!db || !secrets.available()) return;
  let changed = false;
  for (const domain of Object.keys(db)) {
    const entry = db[domain];
    if (entry && typeof entry.password === 'string' && entry.password && !secrets.isEncrypted(entry.password)) {
      entry.password = secrets.encrypt(entry.password);
      changed = true;
    }
  }
  if (changed) { writeAutofill(db); console.log('[Kortex] palavras-passe migradas para armazenamento cifrado'); }
}

async function askPermission(origin, permission, details, k, store, persist) {

// persist=false (separadores privados): as decisÃµes vivem sÃ³ em memÃ³ria e nunca herdam as guardadas.
function setupPermissions(ses, { persist = true } = {}) {
  const store = persist ? (perms = perms || readJson('permissions.json', {})) : {};
  const pendingHere = new Map();
  const key = (origin, p) => origin + '|' + p;
  ses.setPermissionCheckHandler((wc, permission, origin) => AUTO_ALLOW.has(permission) || store[key(origin, permission)] === true);
  ses.setPermissionRequestHandler(async (wc, permission, callback, details) => {
    if (AUTO_ALLOW.has(permission)) return callback(true);
    if (!ASK.has(permission)) return callback(false);
    let origin = '';
    try { origin = new URL(details.requestingUrl || wc.getURL()).origin; } catch {}
    if (!origin || origin === 'null') return callback(false);
    const k = key(origin, permission);
    if (k in store) return callback(store[k]);
    if (pendingHere.has(k)) return pendingHere.get(k).then(callback);
    const p = askPermission(origin, permission, details, k, store, persist);
    pendingHere.set(k, p);
    let allowed = false;
    try { allowed = await p; } finally { pendingHere.delete(k); }
    callback(allowed);
  });
}

function attach(wc) {
  wc.on('before-input-event', (event, input) => {
    if (input.type !== 'keyDown') return;
    const ctrl = input.control || input.meta;
    const key = (input.key || '').toLowerCase();
    const run = (fn) => { event.preventDefault(); try { fn(); } catch (e) { console.error('[Kortex] atalho:', e); } };
    const v = active();
    const awc = v && v.webContents;

    if (key === 'escape') { toChrome('find:close'); return; }
    if (key === 'f12' || (ctrl && input.shift && key === 'i')) return run(() => awc && awc.toggleDevTools());
    if (key === 'f11') return run(() => { const w = ctx.getMainWindow(); w.setFullScreen(!w.isFullScreen()); });
    if (key === 'f5') return run(() => awc && awc.reload());
    if (input.alt && !ctrl && (key === 'arrowleft' || key === 'arrowright')) {
      return run(() => {
        if (!awc) return;
        const h = navOf(awc);
        if (key === 'arrowleft') { if (h.canGoBack()) h.goBack(); } else if (h.canGoForward()) h.goForward();
      });
    }
    if (!ctrl) return;

    if (key === '+' || key === '=') return run(() => zoom(0.1));
    if (key === '-') return run(() => zoom(-0.1));
    if (key === '0') return run(() => zoom(0, true));

    if (input.shift) {
      if (key === 't') return run(reopenClosed);
      if (key === 'n') return run(() => ctx.createNewTab('about:blank', true));
      if (key === 'a') return run(() => toChrome('assistant:toggle'));
      if (key === 'k') return run(() => { toChrome('tabsearch:open'); ctx.getChromeView().webContents.focus(); });
      if (key === 'b') return run(() => toChrome('bookmarks:toggle-bar'));
      if (key === 'r') return run(() => awc && awc.reloadIgnoringCache());
      if (key === 'tab') return run(() => cycle(-1));
      return;
    }

    if (key === 't') return run(() => ctx.createNewTab());
    if (key === 'w') return run(() => ctx.closeTab(ctx.getActiveId()));
    if (key === 'tab') return run(() => cycle(1));
    if (key === 'r') return run(() => awc && awc.reload());

function schemeOf(url) {
  const m = /^([a-z][a-z0-9+.-]*):/i.exec(String(url || '').trim());

function classify(url) {
  const u = String(url || '').trim();
